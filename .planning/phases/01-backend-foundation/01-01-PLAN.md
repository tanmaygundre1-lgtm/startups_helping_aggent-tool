Create backend/middleware/validationMiddleware.js with reusable validators:

```javascript
const { body, param, query, validationResult } = require('express-validator');

module.exports = {
  validateProfileCreation: [
    body('profileType').isIn(['founder', 'candidate']).withMessage('Invalid profile type'),
    body('yearsExperience').if(b => b.profileType === 'founder').isInt({ min: 0 }).withMessage('Experience must be >= 0'),
    body('skills').if(b => b.profileType === 'candidate').isArray({ min: 1 }).withMessage('Skills must be non-empty array'),
    handleValidationErrors
  ],
  
  validateIdeaCreation: [
    body('title').isString().trim().isLength({ min: 3, max: 200 }).withMessage('Title 3-200 chars'),
    body('description').isString().trim().isLength({ min: 20 }).withMessage('Description at least 20 chars'),
    body('stage').isIn(['concept', 'mvp', 'launched']).withMessage('Invalid stage'),
    body('requiredSkills').isArray({ min: 1 }).withMessage('At least 1 required skill'),
    handleValidationErrors
  ],
  
  validateIdParam: [
    param('id').isMongoId().withMessage('Invalid ID format'),
    handleValidationErrors
  ],
  
  handleValidationErrors: (req, res, next) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ error: 'Validation failed', details: errors.array() });
    }
    next();
  }
};
```

- POST /api/auth/profile: validateProfileCreation

Standardize error responses:

- Include try-catch in all async controllers to catch unhandled errors and return 500

**2. backend/tests/e2e.test.js** (Jest + Supertest template):

```javascript
const request = require('supertest');
const app = require('../server');
const mongoose = require('mongoose');
const User = require('../models/User');

describe('Phase 1 E2E Tests', () => {
  let founderToken = 'test-founder-uid';
  let ideaId;

  beforeAll(async () => {
    // Connect to test DB
    await mongoose.connect(process.env.MONGO_TEST_URI);
  });

  afterAll(async () => {
    await User.deleteMany({ email: 'e2e@test.com' });
    await mongoose.disconnect();
  });

  describe('Founder Profile', () => {
    it('POST /api/auth/profile creates founder profile', async () => {
      const res = await request(app)
        .post('/api/auth/profile')
        .set('Authorization', `Bearer ${founderToken}`)
        .send({
          profileType: 'founder',
          companyStage: 'idea',
          expertiseAreas: ['AI'],
          lookingFor: ['Backend'],
          yearsExperience: 5
        });

      expect(res.status).toBe(201);
      expect(res.body.userId).toBeDefined();
    });

    it('GET /api/auth/profile returns founder profile', async () => {
      const res = await request(app)
        .get('/api/auth/profile')
        .set('Authorization', `Bearer ${founderToken}`);

      expect(res.status).toBe(200);
      expect(res.body.profileType).toBe('founder');
    });
  });

  describe('Idea Management', () => {
    it('POST /api/ideas creates new idea', async () => {
      const res = await request(app)
        .post('/api/ideas')
        .set('Authorization', `Bearer ${founderToken}`)
        .send({
          title: 'E2E Test Idea',
          description: 'This is a test idea with substantial description to pass validation',
          stage: 'concept',
          targetMarket: 'Tech',
          requiredSkills: ['Node.js'],
          requiredRoles: ['Backend'],
          fundingStage: 'bootstrapped'
        });

      expect(res.status).toBe(201);
      ideaId = res.body.ideaId;
    });

    it('POST /api/ai/analyze generates analysis', async () => {
      const res = await request(app)
        .post('/api/ai/analyze')
        .set('Authorization', `Bearer ${founderToken}`)
        .send({ ideaId });

      expect(res.status).toBe(200);
      expect(res.body.extractedRoles).toBeDefined();
      expect(Array.isArray(res.body.extractedSkills)).toBe(true);
    });
  });
});
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → API | Untrusted input from user via HTTP requests |
| API → Firebase Auth | Verify JWT tokens, trust Firebase only for authenticated user ID |
| API → MongoDB | Parameterized queries to prevent injection |
| API → Vercel AI | Output from AI model treated as untrusted, parse with JSON schema validation |
| Founder → Candidate Data | Candidate profiles are semi-public (discoverable), but email/contact hidden from non-matching |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-Spoofing | Spoofing | API endpoints | mitigate | Firebase Auth middleware validates JWT on all protected routes; req.user.uid is cryptographically signed |
| T-01-Tampering | Tampering | Profile data | mitigate | Request validation middleware (express-validator) sanitizes input; use parameterized Mongoose queries to prevent NoSQL injection |
| T-01-Tampering | Tampering | AI analysis output | mitigate | Parse AI response with JSON.parse inside try-catch; validate structure (check for required keys: roles, skills, techStack) before storing |
| T-01-Repudiation | Repudiation | Idea edits | mitigate | Log all create/update operations with timestamp and userId; add updatedBy field to track who changed what (Phase 2) |
| T-01-InfoDisclosure | Information Disclosure | Candidate profiles | accept | Candidate names/skills public during discovery phase; emails never exposed; private user data (payment, SSN) not collected in Phase 1 |
| T-01-Denial | Denial | AI endpoint | mitigate | Add rate limiting: max 10 analyses per founder per hour (use Redis or in-memory counter); return 429 if exceeded |
| T-01-Elevation | Elevation | Non-founder API access | mitigate | POST /api/ideas checks profileType == 'founder' from User model (not just request body); route guards use req.user.uid from Firebase |
| T-01-SC | Supply Chain | npm packages | mitigate | Lock versions in package-lock.json; `npm audit` before Phase 2 commit; block `ai`, `@ai-sdk/*`, `express-validator` with `[ASSUMED]` legitimacy until manual verification |

## Security Implementation Details

- **Authentication:** Firebase JWT validated by authMiddleware on all protected routes; JWT decoded and req.user.uid set
- **Input Validation:** express-validator middleware on all POST/PUT routes; reject if validation fails with 400 before controller executes
- **Database Safety:** Use Mongoose parameterized queries only; never string-interpolate user input into queries
- **AI Response Parsing:** Wrap JSON.parse in try-catch; log parse errors; return 500 if AI response invalid
- **Error Messages:** Never leak stack traces or database schema in error responses; use generic "Server error" for 500
- **Logging:** Log auth failures, validation failures, and DB errors to console (implement structured logging in Phase 2)
- **Rate Limiting:** Per-user rate limiting on AI analysis endpoint (Task 5 mentions caching for 1 hour as simple mitigation)
